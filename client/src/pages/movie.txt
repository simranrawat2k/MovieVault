const genres = [
  "All Movies",
  "Action",
  "Adventure",
  "Romance",
  "Sci-Fi",
  "Thriller",
];

function Movies() {
  const [search, setSearch] = useState("");
  const [activeGenre, setActiveGenre] = useState("All Movies");
  const [savedMovies, setSavedMovies] = useState([]);
  const [showSavedOnly, setShowSavedOnly] = useState(false);
  const [currentPage, setCurrentPage] = useState(1);

  const moviesPerPage = 8;

  const filteredMovies = useMemo(() => {
    return moviesData.filter((movie) => {
      const matchesSearch =
        movie.title.toLowerCase().includes(search.toLowerCase()) ||
        movie.genre.toLowerCase().includes(search.toLowerCase());

      const matchesGenre =
        activeGenre === "All Movies" || movie.genre === activeGenre;

      const matchesSaved =
        !showSavedOnly || savedMovies.includes(movie.id);

      return matchesSearch && matchesGenre && matchesSaved;
    });
  }, [search, activeGenre, showSavedOnly, savedMovies]);

  const totalPages = Math.ceil(filteredMovies.length / moviesPerPage);

  const visibleMovies = filteredMovies.slice(
    (currentPage - 1) * moviesPerPage,
    currentPage * moviesPerPage
  );

  const toggleSaved = (movieId) => {
    setSavedMovies((previous) =>
      previous.includes(movieId)
        ? previous.filter((id) => id !== movieId)
        : [...previous, movieId]
    );
  };

  const changeGenre = (genre) => {
    setActiveGenre(genre);
    setCurrentPage(1);
  };

  const changeSearch = (value) => {
    setSearch(value);
    setCurrentPage(1);
  };

  return (
    <Page>
      <Navbar>
        <NavInner>
          <Brand to="/movies">
            <Logo>
              <FaFilm />
            </Logo>
            <BrandName>
              Movie<span>Vault</span>
            </BrandName>
          </Brand>

          <NavLinks>
            <NavLink $active to="/movies">
              Discover
            </NavLink>

            <NavButton
              $active={showSavedOnly}
              onClick={() => {
                setShowSavedOnly(!showSavedOnly);
                setCurrentPage(1);
              }}
            >
              <FaBookmark />
              My Collection
              {savedMovies.length > 0 && (
                <SavedCount>{savedMovies.length}</SavedCount>
              )}
            </NavButton>
          </NavLinks>

          <ProfileIcon title="Account">
            <FaUserCircle />
          </ProfileIcon>
        </NavInner>
      </Navbar>

      <Main>
        <Hero>
          <HeroContent>
            <Eyebrow>
              <FaPlay /> YOUR NEXT MOVIE NIGHT STARTS HERE
            </Eyebrow>

            <HeroTitle>
              Discover your next
              <GradientText> favourite film.</GradientText>
            </HeroTitle>

            <HeroDescription>
              Explore movies, find hidden gems, and keep your favourites
              all in one place.
            </HeroDescription>

            <SearchBox>
              <FaSearch />
              <SearchInput
                type="search"
                placeholder="Search movies or genres..."
                value={search}
                onChange={(e) => changeSearch(e.target.value)}
              />
              {search && (
                <ClearButton onClick={() => changeSearch("")}>
                  Clear
                </ClearButton>
              )}
            </SearchBox>
          </HeroContent>

          <HeroDecoration>
            <DecorationCircle />
            <DecorationFilm>
              <FaFilm />
            </DecorationFilm>
            <DecorationText>YOUR CINEMA. YOUR WORLD.</DecorationText>
          </HeroDecoration>
        </Hero>

        <SectionHeader>
          <div>
            <SectionTitle>
              {showSavedOnly ? "My Collection" : "Explore Movies"}
            </SectionTitle>
            <SectionDescription>
              {filteredMovies.length}{" "}
              {filteredMovies.length === 1 ? "movie" : "movies"} to discover
            </SectionDescription>
          </div>
        </SectionHeader>

        <GenreList>
          {genres.map((genre) => (
            <GenreButton
              key={genre}
              $active={activeGenre === genre}
              onClick={() => changeGenre(genre)}
            >
              {genre}
            </GenreButton>
          ))}
        </GenreList>

        {visibleMovies.length > 0 ? (
          <MovieGrid>
            {visibleMovies.map((movie) => {
              const isSaved = savedMovies.includes(movie.id);

              return (
                <MovieCard key={movie.id}>
                  <PosterWrapper>
                    <Poster
                      src={movie.poster}
                      alt={`${movie.title} poster`}
                      loading="lazy"
                    />

                    <PosterOverlay />

                    <Rating>
                      <FaStar />
                      {movie.rating}
                    </Rating>

                    <SaveButton
                      $saved={isSaved}
                      title={isSaved ? "Remove from collection" : "Save movie"}
                      aria-label={
                        isSaved ? "Remove saved movie" : "Save movie"
                      }
                      onClick={() => toggleSaved(movie.id)}
                    >
                      <FaHeart />
                    </SaveButton>

                    <PosterBottom>
                      <GenreTag>{movie.genre}</GenreTag>
                    </PosterBottom>
                  </PosterWrapper>

                  <MovieInfo>
                    <MovieTitle>{movie.title}</MovieTitle>

                    <MovieMeta>
                      <span>{movie.year}</span>
                      <MetaDot />
                      <span>{movie.genre}</span>
                    </MovieMeta>

                    <MovieDescription>
                      {movie.description}
                    </MovieDescription>
                  </MovieInfo>
                </MovieCard>
              );
            })}
          </MovieGrid>
        ) : (
          <EmptyState>
            <EmptyIcon>
              <FaFilm />
            </EmptyIcon>
            <EmptyTitle>
              {showSavedOnly ? "Your collection is empty" : "No movies found"}
            </EmptyTitle>
            <EmptyDescription>
              {showSavedOnly
                ? "Save movies with the heart button and find them here."
                : "Try another search or choose a different genre."}
            </EmptyDescription>

            <ResetButton
              onClick={() => {
                setSearch("");
                setActiveGenre("All Movies");
                setShowSavedOnly(false);
                setCurrentPage(1);
              }}
            >
              Explore all movies
            </ResetButton>
          </EmptyState>
        )}

        {totalPages > 1 && (
          <Pagination>
            <PageButton
              aria-label="Previous page"
              disabled={currentPage === 1}
              onClick={() => setCurrentPage((page) => page - 1)}
            >
              <FaChevronLeft />
            </PageButton>

            {Array.from({ length: totalPages }, (_, index) => index + 1).map(
              (page) => (
                <PageButton
                  key={page}
                  $active={currentPage === page}
                  onClick={() => setCurrentPage(page)}
                >
                  {page}
                </PageButton>
              )
            )}

            <PageButton
              aria-label="Next page"
              disabled={currentPage === totalPages}
              onClick={() => setCurrentPage((page) => page + 1)}
            >
              <FaChevronRight />
            </PageButton>
          </Pagination>
        )}
      </Main>

      <Footer>
        <Brand to="/movies">
          <Logo>
            <FaFilm />
          </Logo>
          <BrandName>
            Movie<span>Vault</span>
          </BrandName>
        </Brand>
        <FooterText>Find a story worth staying up for.</FooterText>
      </Footer>
    </Page>
  );
}

export default Movies;
